================================================================================
PROJECT: Texas Hill Country Hemp Finder (THC Hemp Finder)
================================================================================
OVERVIEW:
A map‑based locator for 21+ adults in the Texas Hill Country. Businesses can list
multiple locations, select the brands they carry, and upload coupons & photos.
Users search by location, category, or brand, view business details, and download
coupons. Admin approves/rejects listings with personalized messages, toggles
featured shops, manages a site‑wide banner ad, a pop‑up ad, and a featured brands
program. Monetization handled entirely offline. Slick, street‑culture‑meets‑Texas
design with animations.

================================================================================
TECH STACK
================================================================================
Frontend:
  - React 18 (Vite)
  - React Router v6
  - React‑Leaflet v4 + Leaflet v1.9
  - Axios
  - CSS3 (custom, no framework)

Backend:
  - Node.js 20 + Express 4
  - better‑sqlite3 (SQLite)
  - express‑session + connect‑sqlite3 (session store)
  - bcryptjs
  - multer (file uploads)
  - node‑fetch (for geocoding)
  - CORS

Database:
  - SQLite (file: database.sqlite)

Fonts (loaded via Google Fonts):
  - Permanent Marker (headings)
  - Montserrat (body)

No paid APIs or external services required.

function requireLogin(req, res, next) {
  if (req.session && req.session.userId) {
    return next();
  }
  return res.status(401).json({ error: 'Authentication required' });
}

function requireAdmin(req, res, next) {
  if (req.session && req.session.role === 'admin') {
    return next();
  }
  return res.status(403).json({ error: 'Admin access required' });
}

function requireBusiness(req, res, next) {
  if (req.session && req.session.role === 'business') {
    return next();
  }
  return res.status(403).json({ error: 'Business account required' });
}

module.exports = { requireLogin, requireAdmin, requireBusiness };

const express = require('express');

const db = require('../database');
const router = express.Router();

router.post('/register', (req, res) => {
  const { email, password } = req.body;
  if (!email || !password) return res.status(400).json({ error: 'Email and password required' });
  const existing = db.prepare('SELECT id FROM users WHERE email = ?').get(email);
  if (existing) return res.status(400).json({ error: 'Email already in use' });
  const hash = bcrypt.hashSync(password, 10);
  const result = db.prepare('INSERT INTO users (email, password_hash, role) VALUES (?, ?, ?)')
    .run(email, hash, 'business');
  req.session.userId = result.lastInsertRowid;
  req.session.role = 'business';
  return res.json({ success: true, userId: result.lastInsertRowid });
});

router.post('/login', (req, res) => {
  const { email, password } = req.body;
  if (!email || !password) return res.status(400).json({ error: 'Email and password required' });
  const user = db.prepare('SELECT * FROM users WHERE email = ?').get(email);
  if (!user || !bcrypt.compareSync(password, user.password_hash))
    return res.status(401).json({ error: 'Invalid credentials' });
  req.session.userId = user.id;
  req.session.role = user.role;
  return res.json({ success: true, userId: user.id, role: user.role });
});

router.post('/logout', (req, res) => {
  req.session.destroy(err => {
    if (err) return res.status(500).json({ error: 'Logout failed' });
    res.clearCookie('connect.sid');
    return res.json({ success: true });
  });
});

router.get('/me', (req, res) => {
  if (!req.session.userId) return res.status(401).json({ error: 'Not logged in' });
  const user = db.prepare('SELECT id, email, role FROM users WHERE id = ?').get(req.session.userId);
  if (!user) {
    req.session.destroy(() => {});
    return res.status(401).json({ error: 'User not found' });
  }
  return res.json(user);
});

module.exports = router;

async function geocode(address) {
  try {
    const url = `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(address)}`;
    const response = await fetch(url, { headers: { 'User‑Agent': 'THCFinder/1.0' } });
    const data = await response.json();
    if (data && data.length > 0) return { lat: parseFloat(data[0].lat), lng: parseFloat(data[0].lon) };
  } catch (e) { console.error('Geocoding error:', e); }
  return null;
}

// Haversine helper
function haversineDistance(lat1, lon1, lat2, lon2) {
  const R = 6371;
  const dLat = (lat2 ‑ lat1) * Math.PI / 180;
  const dLon = (lon2 ‑ lon1) * Math.PI / 180;
  const a = Math.sin(dLat / 2) ** 2 +
            Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
            Math.sin(dLon / 2) ** 2;
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 ‑ a));
}

// Public: get approved businesses (with filters)
router.get('/', (req, res) => {
  const { search, lat, lng, radius, category, brand, sort, featured } = req.query;
  let query = `
    SELECT b.*, GROUP_CONCAT(bc.category) as categories
    FROM businesses b
    LEFT JOIN business_categories bc ON b.id = bc.business_id
  `;
  const conditions = ['b.status = "approved"'];
  const params = [];

  if (featured === '1') conditions.push('b.is_featured = 1');
  if (search) {
    conditions.push('(b.name LIKE ? OR b.address LIKE ?)');
    params.push(`%${search}%`, `%${search}%`);
  }
  query += ' WHERE ' + conditions.join(' AND ');
  query += ' GROUP BY b.id';

  let businesses = db.prepare(query).all(...params);

  // Brand filter (post‑process if brand param present)
  if (brand) {
    const brandId = db.prepare('SELECT id FROM brands WHERE name = ?').get(brand);
    if (brandId) {
      businesses = businesses.filter(b => {
        const bizBrands = db.prepare('SELECT brand_id FROM business_brands WHERE business_id = ?').all(b.id).map(r => r.brand_id);
        return bizBrands.includes(brandId.id);
      });
    }
  }

  // Category filtering (if multiple, require all)
  if (category) {
    const cats = category.split(',');
    businesses = businesses.filter(b => {
      const bizCats = b.categories ? b.categories.split(',') : [];
      return cats.every(c => bizCats.includes(c));
    });
  }

  // Distance filtering
  if (lat && lng && radius) {
    const centerLat = parseFloat(lat);
    const centerLng = parseFloat(lng);
    const rad = parseFloat(radius);
    businesses = businesses.filter(b => {
      if (b.lat == null || b.lng == null) return false;
      const dist = haversineDistance(centerLat, centerLng, b.lat, b.lng);
      b.distance = dist;
      return dist <= rad;
    });
  }

  // Sort
  if (sort === 'distance' && lat && lng) {
    businesses.sort((a, b) => (a.distance || Infinity) ‑ (b.distance || Infinity));
  } else {
    businesses.sort((a, b) => b.is_featured ‑ a.is_featured || b.id ‑ a.id);
  }

  // Enrich with brands
  businesses = businesses.map(b => {
    const bizBrands = db.prepare(`
      SELECT br.id, br.name, br.is_featured FROM business_brands bb
      JOIN brands br ON br.id = bb.brand_id WHERE bb.business_id = ?
    `).all(b.id);
    return {
      ...b,
      categories: b.categories ? b.categories.split(',') : [],
      brands: bizBrands
    };
  });

  res.json(businesses);
});

router.post('/', requireLogin, requireBusiness, async (req, res) => {
  const { name, address, phone, website, hours, description, categories, brand_ids } = req.body;
  if (!name || !address) return res.status(400).json({ error: 'Name and address required' });
  const coords = await geocode(address);
  const result = db.prepare(`
    INSERT INTO businesses (owner_id, name, address, lat, lng, phone, website, hours, description)
    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
  `).run(req.session.userId, name, address, coords?.lat || null, coords?.lng || null,
         phone, website, hours, description);
  const businessId = result.lastInsertRowid;
  // Categories
  if (categories && Array.isArray(categories)) {
    const insertCat = db.prepare('INSERT INTO business_categories (business_id, category) VALUES (?, ?)');
    categories.forEach(cat => insertCat.run(businessId, cat));
  }
  // Brands
  if (brand_ids && Array.isArray(brand_ids)) {
    const insertBrand = db.prepare('INSERT INTO business_brands (business_id, brand_id) VALUES (?, ?)');
    brand_ids.forEach(bid => insertBrand.run(businessId, bid));
  }
  res.status(201).json({ id: businessId });
});

// Update business (owner only)
router.put('/:id', requireLogin, requireBusiness, async (req, res) => {
  const business = db.prepare('SELECT * FROM businesses WHERE id = ?').get(req.params.id);
  if (!business || business.owner_id !== req.session.userId) return res.status(403).json({ error: 'Forbidden' });
  const { name, address, phone, website, hours, description, categories, brand_ids } = req.body;
  let coords = null;
  if (address && address !== business.address) coords = await geocode(address);
  db.prepare(`
    UPDATE businesses SET name=?, address=?, lat=COALESCE(?,lat), lng=COALESCE(?,lng),
    phone=?, website=?, hours=?, description=? WHERE id=?
  `).run(
    name || business.name, address || business.address, coords?.lat || null, coords?.lng || null,
    phone !== undefined ? phone : business.phone, website !== undefined ? website : business.website,
    hours !== undefined ? hours : business.hours, description !== undefined ? description : business.description,
    business.id
  );
  // Categories
  if (categories && Array.isArray(categories)) {
    db.prepare('DELETE FROM business_categories WHERE business_id = ?').run(business.id);
    const insertCat = db.prepare('INSERT INTO business_categories (business_id, category) VALUES (?, ?)');
    categories.forEach(cat => insertCat.run(business.id, cat));
  }
  // Brands
  if (brand_ids && Array.isArray(brand_ids)) {
    db.prepare('DELETE FROM business_brands WHERE business_id = ?').run(business.id);
    const insertBrand = db.prepare('INSERT INTO business_brands (business_id, brand_id) VALUES (?, ?)');
    brand_ids.forEach(bid => insertBrand.run(business.id, bid));
  }
  // If resubmitting after rejection, reset status to pending
  if (business.status === 'rejected') {
    db.prepare('UPDATE businesses SET status = "pending", rejection_reason = NULL WHERE id = ?').run(business.id);
  }
  res.json({ success: true });
});

import { useState, useEffect, createContext, useContext } from 'react';
import { getMe } from '../api';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    getMe().then(res => setUser(res.data)).catch(() => setUser(null)).finally(() => setLoading(false));
  }, []);

  const loginUser = (userData) => setUser(userData);
  const logoutUser = () => setUser(null);

  return (
    <AuthContext.Provider value={{ user, loading, loginUser, logoutUser }}>
      {children}
    </AuthContext.Provider>
  );
}

export const useAuth = () => useContext(AuthContext);

function MapCenterUpdater({ center }) {
  const map = useMap();
  useEffect(() => { if (center) map.flyTo(center, map.getZoom()); }, [center, map]);
  return null;
}

import { useState, useEffect } from 'react';
import { getBusinesses } from '../api';
import SearchBar from '../components/SearchBar';
import FilterPanel from '../components/FilterPanel';
import MapView from '../components/MapView';
import BusinessCard from '../components/BusinessCard';
import BannerAd from '../components/BannerAd';

export default function HomePage() {
  const [businesses, setBusinesses] = useState([]);
  const [filtered, setFiltered] = useState([]);
  const [selectedCategories, setSelectedCategories] = useState([]);
  const [brandQuery, setBrandQuery] = useState('');
  const [searchCoords, setSearchCoords] = useState(null);

  useEffect(() => {
    getBusinesses().then(res => {
      setBusinesses(res.data);
      setFiltered(res.data);
    });
  }, []);

  // Filtering effect
  useEffect(() => {
    let list = businesses;
    if (selectedCategories.length > 0) {
      list = list.filter(b => selectedCategories.every(cat => b.categories?.includes(cat)));
    }
    if (brandQuery.trim() !== '') {
      const q = brandQuery.toLowerCase();
      list = list.filter(b => b.brands?.some(br => br.name.toLowerCase().includes(q)));
    }
    setFiltered(list);
  }, [selectedCategories, brandQuery, businesses]);

  const handleSearch = async (query) => {
    try {
      const res = await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}`);
      const data = await res.json();
      if (data.length) setSearchCoords([parseFloat(data[0].lat), parseFloat(data[0].lon)]);
      else alert('Location not found');
    } catch (e) { console.error(e); }
  };
  const handleLocate = () => {
    if (navigator.geolocation) {
      navigator.geolocation.getCurrentPosition(
        pos => setSearchCoords([pos.coords.latitude, pos.coords.longitude]),
        () => alert('Unable to retrieve location')
      );
    } else alert('Geolocation not supported');
  };

  return (
    <div className="home‑page fade‑in">
      <BannerAd />
      <div className="container">
        <div className="search‑area">
          <SearchBar onSearch={handleSearch} onLocate={handleLocate} />
          <FilterPanel
            selectedCategories={selectedCategories}
            onCategoriesChange={setSelectedCategories}
            brandQuery={brandQuery}
            onBrandQueryChange={setBrandQuery}
          />
        </div>
        <div className="content‑layout">
          <div className="map‑container">
            <MapView businesses={filtered} center={searchCoords} />
          </div>
          <div className="list‑container">
            <h2>Results ({filtered.length})</h2>
            {filtered.map(b => <BusinessCard key={b.id} business={b} />)}
          </div>
        </div>
      </div>
      <style>{`
        .home‑page { min‑height: calc(100vh ‑ 70px); }
        .search‑area { padding: 20px 0; }
        .content‑layout { display: flex; gap: 20px; margin‑top: 20px; }
        .map‑container { flex: 1.5; height: 70vh; border‑radius: 10px; overflow: hidden; box‑shadow: 0 4px 20px rgba(0,0,0,0.2); }
        .list‑container { flex: 1; overflow‑y: auto; max‑height: 70vh; }
        @media (max‑width: 768px) {
          .content‑layout { flex‑direction: column; }
          .map‑container, .list‑container { height: 50vh; }
        }
      `}</style>
    </div>
  );
}

import { useState } from 'react';
import { useNavigate, Link } from 'react‑router‑dom';
import { login } from '../api';
import { useAuth } from '../hooks/useAuth';

export default function LoginPage() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const { loginUser } = useAuth();
  const navigate = useNavigate();

  const handleSubmit = async (e) => {
    e.preventDefault();
    try {
      const res = await login(email, password);
      loginUser({ id: res.data.userId, role: res.data.role });
      navigate('/dashboard');
    } catch (err) { setError(err.response?.data?.error || 'Login failed'); }
  };
  return (
    <div className="auth‑page container fade‑in">
      <h2>Login</h2>
      <form onSubmit={handleSubmit}>
        <input type="email" placeholder="Email" value={email} onChange={e => setEmail(e.target.value)} required />
        <input type="password" placeholder="Password" value={password} onChange={e => setPassword(e.target.value)} required />
        {error && <p className="error">{error}</p>}
        <button type="submit" className="btn btn‑primary">Login</button>
      </form>
      <p>Don't have an account? <Link to="/register">Register</Link></p>
